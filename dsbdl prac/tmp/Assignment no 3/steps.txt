> row format delimited
> fields terminated by ','
> lines terminated by '\n'
> stored as textfile;

//  Join operation by creating another table
create table n_flight (fno int, year int, source varchar(10))
> row format delimited
> fields terminated by ','
> lines terminated by '\n'
> stored as textfile;


insert into n_flight values(8343,2003,"pune");


// load data from a text file
load data local inpath "n_flight_data.txt"
> overwrite into table n_flight;


// no new table created , only the join displayed
select a.fno, a.year, a.dest, a.delay, b.source
> from flight a join n_flight b
> on (a.fno = b.fno);   // Primary key



#// Creating index table on "flight(on what field index created)"
